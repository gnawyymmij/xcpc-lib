\subsection{AC 自动机 (Aho-Corasick)}
小写字母稠密转移表。插入返回终点编号，支持重复模式；match 返回每个节点所代表字符串在文本中的出现次数。每组先 clear，插完全部模式后 build，之后才可匹配。构建 $O(26V)$，单次匹配 $O(|text|+V)$。
\begin{lstlisting}
struct AhoCorasick {
    struct Node{array<int,26> next{};int fail=0;};
    vector<Node> t;
    vector<int> order;
    bool built=false;
    AhoCorasick(){clear();}
    void clear(){t.assign(1,{});order.clear();built=false;}
    int add(const string& s){
        assert(!built&&!s.empty());int u=0;
        for(char c:s){
            int x=c-'a';assert(0<=x&&x<26);
            if(!t[u].next[x]){
                int v=t.size();t.push_back({});t[u].next[x]=v;
            }
            u=t[u].next[x];
        }
        return u;
    }
    void build(){
        assert(!built);queue<int> q;
        for(int v:t[0].next)if(v)q.push(v);
        while(!q.empty()){
            int u=q.front();q.pop();order.push_back(u);
            for(int c=0;c<26;c++){
                int v=t[u].next[c],f=t[u].fail;
                if(v)t[v].fail=t[f].next[c],q.push(v);
                else t[u].next[c]=t[f].next[c];
            }
        }
        built=true;
    }
    vector<ll> match(const string& s)const{
        assert(built);vector<ll> count(t.size());int u=0;
        for(char c:s){int x=c-'a';assert(0<=x&&x<26);u=t[u].next[x];count[u]++;}
        for(int i=int(order.size())-1;i>=0;i--){
            int v=order[i];count[t[v].fail]+=count[v];
        }
        return count;
    }
};
\end{lstlisting}

\paragraph{使用条件与多测} build 会补全所有转移，不能再次 build 或继续 add；下一组必须 clear。空模式未定义且禁止插入。稀疏字符集版本需独立使用真实 trie 边，根的 fail 为 0 或显式跳根特判，不能直接沿用本实现补全后的转移或把 fail[root] 设为 root 后无条件跳 fail，否则可能死循环。
