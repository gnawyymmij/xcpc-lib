vector<ll> Inv(vector<ll> &a){
    if(a.empty()||a[0]==0){
        cerr<<"Err inverse don't exist.\n";
        assert(0);
    }
    int n=a.size();
    vector<ll>R(1,qp(a[0]));

        vector<ll>Q(a.begin(),a.begin()+min(n,k*2));
