\subsubsection{斯坦纳树 (Steiner Tree)}
斯坦纳树 (Steiner Tree)，用于在图中连接指定的一组“关键点”，使得总边权最小。
\begin{lstlisting}
ll n,m,s;
ll dis[110][110];
ll id[110],a[110];
ll dp[110][1500];
bool fl[110];
struct node{ll x,id;};
bool operator <(node a,node b){return a.x>b.x;}
priority_queue<node>q;
ll res=0x3f3f3f3f3f3f3f3f;
int main(){
	n=d,m=d,s=d;res=0x3f3f3f3f3f3f3f3f;
    while(!q.empty())q.pop();fill(id,id+n+1,0);
	memset(dis,0x3f,sizeof(dis));
    f(i,1,n)dis[i][i]=0;
	f(i,1,m){
		ll u=d,v=d,w=d;
		dis[u][v]=dis[v][u]=min(dis[u][v],w);
	}f(i,1,s)id[a[i]=d]=i;
	f(k,1,n)f(i,1,n)f(j,1,n)dis[i][j]=min(dis[i][j],dis[i][k]+dis[k][j]);
	memset(dp,0x3f,sizeof(dp));
	f(i,1,n)dp[i][0]=0;
	f(i,1,s)dp[a[i]][1<<(i-1)]=0;
	f(S,1,(1<<s)-1){
		f(i,1,n)for(int t=S;t;t=(t-1)&S)dp[i][S]=min(dp[i][S],dp[i][t]+dp[i][S^t]);
		f(i,0,n)fl[i]=0;
		dp[0][S]=0;q.push({0,0});
		while(!q.empty()){
			ll u=q.top().id;q.pop();
			if(fl[u])continue;fl[u]=1;
			if(u==0)f(i,1,n)q.push({dp[i][S],i});
			else{
				f(i,1,n)if(dp[u][S]+dis[u][i]<dp[i][S]){
					dp[i][S]=dp[u][S]+dis[u][i];
					q.push({dp[i][S],i});
				}
			}
		}
	}f(i,1,n)res=min(res,dp[i][(1<<s)-1]);
	cout<<res;
	return 0;
}
\end{lstlisting}

\paragraph{多测与注意} 保留 Floyd 加子集 DP；多测已重置 res、队列、id。边权非负，状态数组按终端数开；终端数为 0 时答案另取 0，不可达时按题目输出。
