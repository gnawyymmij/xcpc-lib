\subsection{RMQ (ST 表)}
ST 表 (Sparse Table)，用于 O(1) 查询区间最值 (RMQ)。
\begin{lstlisting}
ll s[100010][20];
ll lg[100010];
ll get(ll l,ll r){
  ll L=r-l+1,Lg=lg[L];
  return max(s[l][Lg],s[r-(1<<Lg)+1][Lg]);
}
int main(){
	n=d,m=d;lg[1]=0;assert(1<=n&&n<=100000);
	f(i,1,n)s[i][0]=d;
	f(j,1,19)f(i,1,n){
		if(i+(1<<(j-1))>n)s[i][j]=s[i][j-1];
		else s[i][j]=max(s[i][j-1],s[i+(1<<(j-1))][j-1]);
	}f(i,2,n)lg[i]=lg[i/2]+1;
}
\end{lstlisting}


\paragraph{使用条件与多测} 建表完整覆盖本次有效区间，令 lg[1]=0 后重算到 n 即可多测复用，无须整表清零。要求 $1\le l\le r\le n\le100000$；预处理 $O(n\log n)$，查询 $O(1)$。重叠区间合并要求幂等性，不能直接替换为求和。
