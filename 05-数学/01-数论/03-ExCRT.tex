\subsubsection{ExCRT (Extended Chinese Remainder Theorem)}
扩展中国剩余定理，用于求解模数不互质的线性同余方程组。
\begin{lstlisting}
ll x,y,c;
void exgcd(ll a,ll b){
	if(b==0){x=1,y=0;return;}
	exgcd(b,a%b);ll xx=x,yy=y;
	x=yy,y=xx-a/b*yy;
}ll gcd(ll a,ll b){
	return (b==0)?a:gcd(b,a%b);
}
ll A,B,a,b;
// merge {x equiv A mod B} and {x equiv a mod b}
bool merge(ll &A,ll &B,ll a,ll b){
    c=a-A;ll g=gcd(b,B);
    if(c%g)return false;
    exgcd(B,b);
    x=(__int128)x*(c/g)%(b/g);b=b/g*B;
    A=((A+(__int128)x*B%b)%b+b)%b;B=b;return true;
}
\end{lstlisting}

\paragraph{多测与注意} 保留 A、B、x、y、c 及原合并流程，merge 仅增加 bool 返回值表示有无解。每组 A=0、B=1 开始；若周期超 ll，按题目改类型。
