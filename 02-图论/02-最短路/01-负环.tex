\subsubsection{负环 (Negative Cycle Detection - SPFA)}
使用 SPFA (Shortest Path Faster Algorithm) 检测图中的负环。
\begin{lstlisting}
ll dis[10010];
bool in[10010];
ll c[10010];
queue<ll>q;
bool work(){
	dis[1]=0;in[1]=1;q.push(1);
	while(!q.empty()){
		ll u=q.front();q.pop();in[u]=0;
		for(int i=hd[u];i;i=e[i].nx){
			ll v=e[i].v;if(dis[v]>dis[u]+e[i].w){
				dis[v]=dis[u]+e[i].w;
				if(!in[v]){
					in[v]=1,c[v]++;
					q.push(v);if(c[v]>n){return 1;}
				}
			}
		}
	}return 0;
}
\end{lstlisting}
