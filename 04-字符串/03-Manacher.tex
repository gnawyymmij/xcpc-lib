\subsection{Manacher}
Manacher，返回最长回文子串长度。
\begin{lstlisting}
int manacher(const string& s){
    vector<int> a(1,256);
    for(unsigned char c:s){a.push_back(c);a.push_back(256);}
    int n=a.size(),mid=0,right=-1,ans=0;
    vector<int> radius(n);
    for(int i=0;i<n;i++){
        if(i<=right)radius[i]=min(right-i,radius[2*mid-i]);
        while(i-radius[i]-1>=0&&i+radius[i]+1<n&&
              a[i-radius[i]-1]==a[i+radius[i]+1])radius[i]++;
        if(i+radius[i]>right)mid=i,right=i+radius[i];
        ans=max(ans,radius[i]);
    }
    return ans;
}
\end{lstlisting}

\paragraph{多测与注意} 局部变量直接多测，返回最长回文长度。时间、空间 $O(n)$。
