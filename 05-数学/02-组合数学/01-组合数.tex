\subsubsection{组合数 (Binomial Coefficients)}
预处理阶乘及其逆元，O(1) 查询组合数。
\begin{lstlisting}
namespace binom{
	const ll Lim=300010,mod=998244353;
	ll jc[Lim],inv[Lim],inc[Lim];
	void pre(){
        static bool ready=false;if(ready)return;ready=true;
		jc[0]=jc[1]=inc[0]=inc[1]=inv[0]=inv[1]=1;
		f(i,2,Lim-1)jc[i]=jc[i-1]*i%mod,inv[i]=(mod-mod/i)*inv[mod%i]%mod,
		inc[i]=inc[i-1]*inv[i]%mod;
	}ll C(ll n,ll m){if(n<0||m<0||n<m)return 0;assert(n<Lim);return jc[n]*inc[m]%mod*inc[n-m]%mod;}
}
\end{lstlisting}


\paragraph{使用条件与多测} pre 在第一次查询前调用一次即可跨组复用；要求素数 mod 且预处理上限小于 mod。查询 $n\ge Lim$ 会越界，已增加断言；大 n 改用 Lucas。变更模数或上限时须重建预处理，不能复用旧逆元。
