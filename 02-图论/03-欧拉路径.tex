\subsection{欧拉路径 (Eulerian Path)}
有向图 Hierholzer，顶点从 1 编号；返回空 vector 表示无解。检查度数与实际使用边数，可处理非连通图、重边、自环；无边且 $n>0$ 返回顶点 1。显式栈避免 $O(m)$ 深递归。局部变量支持多测；排序后输出字典序最小路径，时间 $O(n+m\log m)$，不要求字典序时可去掉排序，降为 $O(n+m)$。
\begin{lstlisting}
vector<int> euler_directed(int n,const vector<pair<int,int>>& edges){
    if(n<=0)return {};
    vector<vector<int>> g(n+1);vector<int> in(n+1),out(n+1),cur(n+1);
    for(auto [u,v]:edges){g[u].push_back(v);out[u]++;in[v]++;}
    int start=-1,finish=-1;
    for(int u=1;u<=n;u++){
        if(out[u]-in[u]==1){if(start!=-1)return {};start=u;}
        else if(in[u]-out[u]==1){if(finish!=-1)return {};finish=u;}
        else if(in[u]!=out[u])return {};
        sort(g[u].begin(),g[u].end());
    }
    if((start==-1)!=(finish==-1))return {};
    if(start==-1){start=1;for(int u=1;u<=n;u++)if(out[u]){start=u;break;}}
    vector<int> st{start},path;
    while(!st.empty()){
        int u=st.back();
        if(cur[u]<int(g[u].size()))st.push_back(g[u][cur[u]++]);
        else path.push_back(u),st.pop_back();
    }
    if(path.size()!=edges.size()+1)return {};
    reverse(path.begin(),path.end());return path;
}
\end{lstlisting}

\paragraph{多测与注意} 局部变量直接多测。无向图需改成共享边编号判重，并判断奇度点为 0 或 2；仍要检查是否用尽所有边。不要求字典序时去掉排序。
