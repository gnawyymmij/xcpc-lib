\subsubsection{树的重心 (Tree Centroid)}
从点分模板贺的，所以有fl数组表示有没有被处理过。
\begin{lstlisting}
ll sz[100010],g,mn;
ll f[100010];
bool fl[100010];
ll siz;
void dfsz(ll u,ll fa){if(fl[u])return;
	f[u]=fa;siz++;
	sz[u]=1;for(int i=hd[u];i;i=e[i].nx){
		ll v=e[i].v,w=e[i].w;
		if(v==fa||fl[v])continue;
		dfsz(v,u),sz[u]+=sz[v];
	}
}
void dfsg(ll u,ll fa){if(fl[u])return;
	ll mx=0;
	if(f[u]!=0)mx=siz-sz[u];
	for(int i=hd[u];i;i=e[i].nx){
		ll v=e[i].v,w=e[i].w;
		if(v==fa||fl[v])continue;
		dfsg(v,u),mx=max(mx,sz[v]);
	}if(mx<mn){mn=mx,g=u;}
}
void getg(ll rt){
	mn=0x7f7f7f7f;siz=0;g=0;
	dfsz(rt,0),dfsg(rt,0);
}
\end{lstlisting}


\paragraph{使用条件与多测} 每组重建邻接表并清 fl[1..n]；同组点分治递归时 fl 是已删重心标记，不能每次 getg 都清空。rt 必须位于当前未删除的树分量；若 fl[rt] 为真返回 g=0。每次 getg 已重置 siz、mn、g，sz/f 在当前分量重新覆盖。深链注意递归栈，单次 $O(\text{分量大小})$。
