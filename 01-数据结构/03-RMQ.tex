\subsection{RMQ (ST 表)}
ST 表 (Sparse Table)，用于 O(1) 查询区间最值 (RMQ)。
\begin{lstlisting}
ll s[100010][20];
ll lg[100010];
ll get(ll l,ll r){
  ll L=r-l+1,Lg=lg[L];
  return max(s[l][Lg],s[r-(1<<Lg)+1][Lg]);
}
int main(){
	n=d,m=d;lg[1]=0;
    f(i,1,n)s[i][0]=d;
    f(i,2,n)lg[i]=lg[i/2]+1;
    f(j,1,lg[n])f(i,1,n-(1LL<<j)+1)
        s[i][j]=max(s[i][j-1],s[i+(1LL<<(j-1))][j-1]);
}
\end{lstlisting}

\paragraph{多测与注意} 多测重读 s[i][0]、重建到 n，lg[1]=0；无需清整表。数组按题目开，第二维开到 $\log_2 n$。预处理 $O(n\log n)$、查询 $O(1)$；重叠合并需幂等，不能直接用于区间和。
