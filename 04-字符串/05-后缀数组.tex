\subsection{后缀数组 (Suffix Array - SA)}
倍增后缀数组，sa 为后缀起点，rank 为排名，lcp[i] 为 sa[i] 与 sa[i-1] 的 LCP。
\begin{lstlisting}
struct SuffixArray {
    vector<int> sa,rank,lcp;
    void build(const string& s){
        int n=s.size();sa.resize(n);rank.resize(n);lcp.assign(n,0);
        if(!n)return;
        vector<int> x(n),y(n),tmp(n),cnt(max(n,256)+1);
        for(int i=0;i<n;i++)x[i]=(unsigned char)s[i]+1,sa[i]=i;
        auto sort_by=[&](const vector<int>& key,int m){
            fill(cnt.begin(),cnt.begin()+m+1,0);
            for(int id:sa)cnt[key[id]]++;
            for(int i=1;i<=m;i++)cnt[i]+=cnt[i-1];
            for(int i=n-1;i>=0;i--)tmp[--cnt[key[sa[i]]]]=sa[i];
            sa.swap(tmp);
        };
        int m=256;
        for(int k=1;;k*=2){
            for(int i=0;i<n;i++)y[i]=(i+k<n?x[i+k]:0);
            sort_by(y,m);sort_by(x,m);
            int classes=1;y[sa[0]]=classes;
            for(int i=1;i<n;i++){
                int a=sa[i-1],b=sa[i];
                if(x[a]!=x[b]||(a+k<n?x[a+k]:0)!=(b+k<n?x[b+k]:0))classes++;
                y[b]=classes;
            }
            x.swap(y);m=classes;if(m==n)break;
        }
        for(int i=0;i<n;i++)rank[sa[i]]=i;
        for(int i=0,h=0;i<n;i++){
            int r=rank[i];if(!r){h=0;continue;}
            int j=sa[r-1];
            while(i+h<n&&j+h<n&&s[i+h]==s[j+h])h++;
            lcp[r]=h;if(h)h--;
        }
    }
};
\end{lstlisting}

\paragraph{多测与注意} 每次 build 重建状态，SA、rank 均为 0 下标，lcp[0]=0。时间 $O(n\log n)$，空间 $O(n)$。
