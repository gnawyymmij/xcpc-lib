\subsubsection{行列式}
如果模数不是质数，需要对行的主元进行辗转相除。
\begin{lstlisting}
ll qp(ll a,ll b){
    ll res=1;
    while(b){
        if(b&1)res=res*a%mod;
        a=a*a%mod,b>>=1;
    }return res;
}
ll det(ll n,ll a[][1010]){
    int sgn=1;
    f(i,1,n)f(j,1,n)a[i][j]=(a[i][j]%mod+mod)%mod;
    f(i,1,n){
        int p=-1;
        f(j,i,n)if(a[j][i]!=0){p=j;break;}
        if(p==-1){return 0;}
        if(p!=i)swap(a[i],a[p]),sgn=-sgn;
        ll iv=qp(a[i][i],mod-2);
        f(j,i+1,n){
            ll o=a[j][i]*iv%mod;
            f(k,i,n)a[j][k]=(a[j][k]-a[i][k]*o%mod+mod)%mod;
        }
    }
    ll res=1;
    f(i,1,n)res=res*a[i][i]%mod;
    if(sgn==-1)res=(mod-res)%mod;
    return res;
}
\end{lstlisting}

\paragraph{使用条件与多测} 代码要求 mod 为质数且主元非零才可用 Fermat 逆元；合数模必须换无除法算法或辗转消元，不能照抄这份代码。每组完整赋矩阵，det 会原地修改它；空矩阵行列式为 1。ll 乘积若可能溢出应转 128 位。时间 $O(n^3+n\log mod)$。
