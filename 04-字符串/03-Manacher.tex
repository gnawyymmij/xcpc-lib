\subsection{Manacher}
Manacher 算法，用于在线性时间内求解字符串的最长回文子串。
\begin{lstlisting}
  m=strlen(C+1);n=0;
	c[++n]='#';
	f(i,1,m)c[++n]=C[i],c[++n]='#';
	f(i,1,n)res[i]=0;
    md=1,r=1;ans=0;
	f(i,2,n){
		if(i>r){
			while(i+res[i]<n&&i-res[i]>1&&c[i+res[i]+1]==c[i-res[i]-1])res[i]++;
			md=i,r=i+res[i];
		}else{
			res[i]=min(r-i,res[md*2-i]);
			while(i+res[i]<n&&i-res[i]>1&&c[i+res[i]+1]==c[i-res[i]-1])res[i]++;
			if(i+res[i]>r)r=i+res[i],md=i;
		}ans=max(ans,res[i]);
	}
	cout<<ans;
\end{lstlisting}

\paragraph{多测与注意} 保留原插井号写法，只补 n、res、ans 的逐组重置。处理串数组开到原串约两倍。
